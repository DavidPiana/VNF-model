\section*{Model Formulation 3}

\subsection*{Sets}
\begin{itemize}
    \item $N$: set of physical nodes
    \item $A$: set of physical arcs
    \item $D$: set of demands
    \item $F$: set of VNF types
    \item $\tau$: virtual sink node ($\tau \notin N$)
    \item $N^+ = N \cup \{\tau\}$: extended node set
    \item $A^+ = A \cup \{(i,\tau) : i \in N\}$: extended arc set (virtual arcs have unlimited capacity)
    \item $O = \{o^k : k \in D\}$: set of origin nodes
\end{itemize}

\subsection*{Parameters}
\begin{itemize}
    \item $u$: capacity of each physical arc $(i,j) \in A$
    \item $q_f$: capacity of service type $f \in F$
    \item $d^k$: bandwidth demand of $k \in D$
    \item $o^k$: origin node of demand $k \in D$ ($o^k \in N$)
    \item $n$: number of services in the chain (same for all demands)
    \item $f: \{1, \dots, n\} \to F$: type of service at position $s$ in the chain
    \item $\alpha_{i,f} \in \{0,1\}$: equals $1$ if service type $f$ can be executed on node $i$ and $i \notin O$, $0$ otherwise
    \item $\tau_{ij} \geq 0$: transport delay on physical arc $(i,j) \in A$ (virtual arcs have zero delay)
    \item $\sigma_f \geq 0$: fixed setup time for VNF type $f \in F$
    \item $\gamma_f \geq 0$: variable processing time per demand for VNF type $f \in F$
    \item $lv_i \in \{1, 2, 3\}$: hierarchical level of node $i \in N$ ($1=EN, 2=CN, 3=BN$)
    \item $c_{batch}$: fixed component of the variable training time for $F2$
    \item $c_{inv}(lv_i)$: level-dependent component of the variable training time for $F2$
    \item $q_{batch}$: fixed component of the setup training time for $F2$
    \item $q_{inv}(lv_i)$: level-dependent component of the setup training time for $F2$
    \item $DS_{\max}$: maximum number of demands that can be served by $F2$ on a single node
    \item $T_{target}$: target maximum training time for the federated learning function $F2$
\end{itemize}

\subsection*{Decision Variables}
\begin{itemize}
    \item $y_{if} \in \{0,1\}$: $1$ if service type $f \in F$ is installed on node $i \in N$
    \item $z_{if}^k \in \{0,1\}$: $1$ if demand $k \in D$ uses service type $f \in F$ on node $i \in N$
    \item $x_{ij}^{ks} \in \{0,1\}$: $1$ if extended arc $(i,j) \in A^+$ is used by demand $k \in D$ in subpath $s \in \{1, \dots, n+1\}$
    \item $w_{uv}^{i, s+1} \in \{0,1\}$: $1$ if extended arc $(u,v) \in A^+$ is used in subpath $s+1$ by the aggregated flow that has just completed service $s$ on node $i$ ($i \in N$, $s \in \{1, \dots, n\}$)
    \item $T_{if} \in \mathbb{R}_+$: total processing time at node $i \in N$ for VNF type $f \in F$
    \item $\Theta_{if}^k \in \mathbb{R}_+$: processing time experienced by demand $k \in D$ when using VNF type $f \in F$ at node $i \in N$
    \item $L^k \in \mathbb{R}_+$: total end-to-end latency for demand $k \in D$ (transport + processing)
    \item $L_{\max} \in \mathbb{R}_+$: maximum latency among all demands
\end{itemize}

\subsection*{Objective Function}
\begin{equation}
    \min L_{\max}
\end{equation}

\subsection*{Constraints}
\begin{align}
    \sum_{i \in N} z_{if}^k &= 1, && \forall k \in D,\ f \in F \label{assign1} \\[4pt]
    z_{if}^k &\leq y_{if}, && \forall k \in D,\ i \in N,\ f \in F : \alpha_{i,f}=1 \label{assign2a} \\[4pt]
    z_{if}^k &= 0, && \forall k \in D,\ i \in N,\ f \in F : \alpha_{i,f}=0 \label{assign2b} \\[4pt]
    y_{if} &= 0, && \forall i \in N,\ f \in F : \alpha_{i,f}=0 \label{assign2c} \\[4pt]
    \sum_{k \in D} z_{if}^{k} &\leq q_f, &&\forall i \in N, f \in F : \alpha_{i,f} = 1 \label{capvnf} \\[4pt]
    \sum_{k \in D} z_{i,F2}^{k} &\leq DS_{\max}, &&\forall i \in N : \alpha_{i,F2} = 1 \label{capF2} \\[4pt]
    \sum_{j:(i,j)\in A^+} x_{ij}^{ks} - \sum_{j:(j,i)\in A} x_{ji}^{ks} &= z_{i,f(s-1)}^k - z_{i,f(s)}^k, && \forall k \in D,\ i \in N,\ s \in \{2,\dots,n\} \label{flowinter} \\[4pt]
    \sum_{j:(i,j)\in A} x_{ij}^{k1} - \sum_{j:(j,i)\in A} x_{ji}^{k1} &= 
    \begin{cases}
        1 - z_{i,f(1)}^k, & i = o^k \\
        - z_{i,f(1)}^k, & i \neq o^k
    \end{cases}, && \forall k \in D,\ i \in N \label{flowfirst} \\[4pt]
    \sum_{j:(i,j)\in A^+} x_{ij}^{k,n+1} - \sum_{j:(j,i)\in A} x_{ji}^{k,n+1} &= z_{i,f(n)}^k, && \forall k \in D,\ i \in N \label{flowlast} \\[4pt]
    \sum_{s=1}^{n+1} \sum_{l:(j,i)\in A} x_{ji}^{ks} &\leq 1, && \forall k \in D,\ i \in N \label{indegree} \\[4pt]
    \sum_{s=1}^{n+1} \sum_{j:(i,j)\in A^+} x_{ij}^{ks} &\leq 1, && \forall k \in D,\ i \in N \label{outdegree} \\[4pt]
    x_{uv}^{k,s+1} &\leq w_{uv}^{i,s+1} + (1 - z_{i,f(s)}^k), && \forall k \in D,\ i \in N,\ (u,v) \in A^+,\ s \in \{1,\dots,n\} \label{coupling} \\[4pt]
    \sum_{v:(u,v)\in A^+} w_{uv}^{i,s+1} &\leq 1, && \forall u \in N,\ i \in N,\ s \in \{1,\dots,n\} \label{nobifurc}
\end{align}

\begin{align}
    T_{if} &= \sigma_f \cdot y_{if} + \gamma_f \cdot \sum_{k \in D} z_{if}^k, && \forall i \in N,\ f \in F \setminus \{F2\} : \alpha_{i,f}=1 \label{proctime_other} \\[4pt]
    T_{i,F2} &= \left(q_{batch} + q_{inv}(lv_i)\right) \cdot y_{i,F2} + \left(c_{batch} + c_{inv}(lv_i)\right) \cdot \sum_{k \in D} z_{i,F2}^k, && \forall i \in N : \alpha_{i,F2}=1 \label{proctime_F2} \\[4pt]
    T_{i,F2} &\leq T_{target}, && \forall i \in N : \alpha_{i,F2}=1 \label{trainingtarget} \\[4pt]
    \Theta_{if}^k &\leq M_f \cdot z_{if}^k, && \forall k \in D,\ i \in N,\ f \in F : \alpha_{i,f}=1 \label{theta1} \\[4pt]
    \Theta_{if}^k &\geq T_{if} - M_f(1 - z_{if}^k), && \forall k \in D,\ i \in N,\ f \in F : \alpha_{i,f}=1 \label{theta2} \\[4pt]
    L^k &= \sum_{s=1}^{n+1} \sum_{(i,j) \in A} \tau_{ij} \cdot x_{ij}^{ks} + \sum_{\substack{i \in N,\ f \in F \\ \alpha_{i,f}=1}} \Theta_{if}^k, && \forall k \in D \label{latency}
\end{align}
\begin{align}
    L_{\max} &\geq L^k, && \forall k \in D \label{maxlatency}
\end{align}
\vspace{1em}

\subsection*{Explanation of Constraints}
\begin{itemize}
    \item[\eqref{assign1}] \textbf{VNF assignment:} each demand must be served by exactly one instance of each required VNF type.
    \item[\eqref{assign2a}--\eqref{assign2c}] \textbf{Feasibility and installation:} a VNF can be used on a node only if it is installed there \eqref{assign2a}, and both usage \eqref{assign2b} and installation \eqref{assign2c} are forbidden on nodes where that VNF type is not executable ($\alpha_{i,f}=0$).
    \item[\eqref{capvnf}] \textbf{VNF capacity:} the total bandwidth processed by service type $f$ on node $i$ cannot exceed $q_f$. Multiplying by $\alpha_{i,f}$ makes the right‑hand side zero when $f$ is not executable on $i$.
    \item[\eqref{capF2}] \textbf{Capacity of F2 (Federated Learning):} the maximum number of demands processed by service $F2$ on node $i$ cannot exceed the limit $DS_{\max}$.
    \item[\eqref{flowinter}] \textbf{Intermediate flow balance:} for subpaths $s = 2,\dots,n$, the net flow entering node $i$ equals the difference between using service $f(s-1)$ and service $f(s)$ at $i$. This links successive services along the chain.
    \item[\eqref{flowfirst}] \textbf{First subpath flow balance:} from the origin $o^k$, one unit of flow is injected; at other nodes the net outflow equals $z_{i,f(1)}^k$, forcing the first VNF to be reached exactly once.
    \item[\eqref{flowlast}] \textbf{Last subpath flow balance:} after the final service $f(n)$, the flow is directed to the virtual sink $\tau$ through arcs $(i,\tau) \in A^+$. The net flow at node $i$ equals $z_{i,f(n)}^k$, meaning the flow leaves the node where the last VNF was processed.
    \item[\eqref{indegree}--\eqref{outdegree}] \textbf{Simple path:} each demand uses at most one incoming and one outgoing arc at any node, preventing cycles and traffic splitting.
    \item[\eqref{coupling}] \textbf{Path unification:} if demand $k$ uses service $s$ on node $i$ ($z_{i,f(s)}^k = 1$), then its subsequent routing ($x_{uv}^{k,s+1}$) must follow the aggregated path $w_{uv}^{i,s+1}$ that originates from $i$. This forces all demands sharing the same service node to travel together toward the sink.
    \item[\eqref{nobifurc}] \textbf{No bifurcation of aggregated paths:} the aggregated path leaving node $i$ after service $s$ cannot branch; at most one outgoing arc per node is used. Together with \eqref{coupling}, this guarantees a single tree rooted at $\tau$ for each service‑stage.
    \item[\eqref{proctime_other}] \textbf{Processing time for generic VNFs:} the total processing time at node $i$ for generic VNF $f$ includes a fixed setup time $\sigma_f$ (paid once if $y_{if}=1$) plus a variable time $\gamma_f$ per demand using that VNF.
    \item[\eqref{proctime_F2}] \textbf{Training time for F2 (Federated Learning):} the processing time for $F2$ depends on the node's hierarchical level $lv_i$. It includes a fixed setup component $\left(q_{batch} + q_{inv}(lv_i)\right)$ plus a variable component $\left(c_{batch} + c_{inv}(lv_i)\right)$ proportional to the number of demands.
    \item[\eqref{trainingtarget}] \textbf{Federated Learning convergence target:} the total training time $T_{i,F2}$ for service $F2$ at node $i$ must not exceed the target time $T_{target}$.
    \item[\eqref{theta1}--\eqref{theta2}] \textbf{Linearization:} $\Theta_{if}^k = T_{if}$ if demand $k$ uses VNF $f$ at node $i$ ($z_{if}^k=1$), otherwise $\Theta_{if}^k = 0$. Here $M_f$ is a sufficiently large constant specific to each VNF type $f$.
    \item[\eqref{latency}] \textbf{End-to-end latency:} total time for demand $k$ is the sum of transport delays on all arcs used plus processing times at all VNFs visited.
    \item[\eqref{maxlatency}] \textbf{Maximum latency:} $L_{\max}$ is forced to be at least as large as the latency of every demand. Minimizing $L_{\max}$ in the objective ensures it equals the maximum latency across all demands.
\end{itemize}

\vspace{1em}
\noindent\textbf{Note on the per-function big-$M_f$ constants:} 
Using a single global large constant for all functions leads to weak LP relaxations. Therefore, we define a specific $M_f$ for each function $f \in F$ to bound $T_{if}$ tightly. A valid choice is:
\begin{align*}
    M_f &= \sigma_f + \gamma_f \cdot |D|, \quad &&\forall f \in F \setminus \{F2\} \\
    M_{F2} &= \min \left( \max_{i \in N} \left( q_{batch} + q_{inv}(lv_i) + (c_{batch} + c_{inv}(lv_i)) \cdot |D| \right), T_{target} \right)
\end{align*}
This bound is tight because the maximum processing time $T_{if}$ for generic functions occurs when all $|D|$ demands use the same VNF instance. For $F2$, the maximum processing time is additionally bounded by the convergence requirement $T_{target}$ from constraint \eqref{trainingtarget}.

\subsection*{Instance Details}

The model has been tested on four different network topologies. Two categories of networks are used:
\begin{itemize}
    \item \textbf{Mandala}: a synthetic hierarchical topology with explicit EN/CN/BN layers, designed to model a realistic 5G MAN (Metropolitan Area Network).
    \item \textbf{SNDLib-based networks} (France, Norway, TA1): real-world topologies from the SNDLib benchmark library, adapted with hierarchical levels and demand sources.
\end{itemize}
